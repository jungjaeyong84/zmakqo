f_clamp(float v, float lo, float hi) => math.max(lo, math.min(hi, v))
f_clamp01(float v) => f_clamp(v, 0.0, 1.0)
f_safe_div(float a, float b) => math.abs(b) > 1e-10 ? a / b : 0.0
f_json_escape(string src) =>
    string txt = str.tostring(src)
    txt := str.replace_all(txt, "\\", "\\\\")
    txt := str.replace_all(txt, '"', '\\"')
    txt := str.replace_all(txt, "\n", "\\n")
    txt := str.replace_all(txt, "\r", "\\r")
    txt := str.replace_all(txt, "\t", "\\t")
    txt
f_json_pair_s(string key, string val) => ',"' + key + '":"' + f_json_escape(val) + '"'
f_json_pair_n(string key, float val) => ',"' + key + '":' + str.tostring(val)
f_json_pair_b(string key, bool val) => ',"' + key + '":' + (val ? "true" : "false")
f_score_color(float score) => score >= core_score_min ? color.lime : score >= early_score_min ? color.yellow : color.orange
f_gate_text(bool pass) => pass ? "PASS" : "BLOCK"
f_append_reason(string current, string reason) => current == "" ? reason : current + " | " + reason

f_shape_text(string direction, string grade) => direction == "LONG" ? (grade == "CORE" ? "LC" : "LE") : (grade == "CORE" ? "SC" : "SE")

f_gate_reason(string side, bool noTradePass, bool htfPass, bool setupPass, bool triggerPass, bool edgePass, string setupType, string triggerType, float score, bool coreRaw, bool earlyRaw) =>
    string reason = ""
    if not noTradePass
        reason := f_append_reason(reason, "NO_TRADE_GATE")
    if not htfPass
        reason := f_append_reason(reason, "HTF_REGIME_BLOCK")
    if not setupPass
        reason := f_append_reason(reason, setupType == "NONE" ? "SETUP_MISSING" : "SETUP_QUALITY_LOW")
    if not triggerPass
        reason := f_append_reason(reason, triggerType == "NONE" ? "TRIGGER_MISSING" : "TRIGGER_NOT_CONFIRMED")
    if not edgePass
