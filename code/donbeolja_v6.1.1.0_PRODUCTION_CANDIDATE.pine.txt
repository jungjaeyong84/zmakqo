f_clamp01(v) => math.max(0.0, math.min(1.0, v))
f_safe_div(a, b) => math.abs(b) > 1e-10 ? a / b : 0.0
f_json_escape(string src) =>
    string txt = str.tostring(src)
    txt := str.replace_all(txt, "\\", "\\\\")
    txt := str.replace_all(txt, '"', '\\"')
    txt := str.replace_all(txt, "\n", "\\n")
    txt := str.replace_all(txt, "\r", "\\r")
    txt := str.replace_all(txt, "\t", "\\t")
    txt
f_json_pair_s(string key, string val) => ",\"" + key + "\":\"" + f_json_escape(val) + "\""
f_json_pair_n(string key, float val) => ",\"" + key + "\":" + str.tostring(val)
f_bool_text(bool v) => v ? "true" : "false"

f_entry_payload(string direction, string grade, float score, string state, string htf, string trigger_type, string risk_mode, float rr, float stop_price, float target_price) =>
    string side = direction == "LONG" ? "BUY" : "SELL"
    string features = "{"
    features += "\"strategy_id\":\"" + STRATEGY_ID + "\""
    features += f_json_pair_s("engine_mode", ENGINE_MODE)
    features += f_json_pair_s("entry_grade", grade)
    features += f_json_pair_s("qty_profile", QTY_PROFILE)
    features += f_json_pair_s("market_state", state)
    features += f_json_pair_s("htf_bias", htf)
    features += f_json_pair_s("trigger_type", trigger_type)
    features += f_json_pair_s("risk_mode", risk_mode)
    features += f_json_pair_n("opportunity_score", score)
    features += f_json_pair_n("rr", rr)
    features += f_json_pair_n("stop_price", stop_price)
    features += f_json_pair_n("target_price", target_price)
    features += f_json_pair_s("_event_intent", "ENTRY")
    features += f_json_pair_s("signal_family", direction)
    features += f_json_pair_s("source_band", grade)
    features += "}"
    string json = "{"
    json += f_json_pair_s("exchange", provider_id)
    json += f_json_pair_s("symbol", syminfo.ticker)
    json += f_json_pair_s("market", syminfo.ticker)
    json += f_json_pair_s("ticker", syminfo.ticker)
    json += f_json_pair_s("tf", timeframe.period)
    json += "\"strategy_id\":\"" + STRATEGY_ID + "\""
    json += f_json_pair_s("engine_mode", ENGINE_MODE)
    json += f_json_pair_s("action", "ENTRY")
    json += f_json_pair_s("event_intent", "ENTRY")
    json += f_json_pair_s("event", direction)
    json += f_json_pair_s("side", side)
    json += f_json_pair_s("direction", direction)
    json += f_json_pair_s("entry_grade", grade)
    json += f_json_pair_s("qty_profile", QTY_PROFILE)
    json += f_json_pair_s("timeframe", timeframe.period)
    json += f_json_pair_s("market_state", state)
    json += f_json_pair_s("htf_bias", htf)
    json += f_json_pair_s("trigger_type", trigger_type)
    json += f_json_pair_s("risk_mode", risk_mode)
    json += f_json_pair_n("qtyPct", webhook_qty_pct)
    json += f_json_pair_n("price", close)
    json += f_json_pair_n("opportunity_score", score)
    json += f_json_pair_n("rr", rr)
    json += f_json_pair_n("stop_price", stop_price)
    json += f_json_pair_n("target_price", target_price)
    json += f_json_pair_n("bar_close_time_utc_ms", time_close)
    json += f_json_pair_n("bar_time", time_close)
    json += ",\"features\":" + features
    json += "}"
    json
