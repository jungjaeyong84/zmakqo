//@version=6
strategy("돈벌자 v5.6.0.2 백테스트", overlay=true,
     default_qty_type=strategy.percent_of_equity, default_qty_value=10,
     initial_capital=14000, currency=currency.USD,
     commission_type=strategy.commission.percent, commission_value=0.04,
     slippage=2, pyramiding=0, process_orders_on_close=true,
     calc_on_every_tick=false)

//─── 설정 ───
grp_entry = "═══ 진입 설정 ═══"
grp_exit  = "═══ 청산 설정 ═══"
grp_filt  = "═══ 필터 ═══"

// 진입 등급 선택
entry_tier = input.string("ALL", "진입 등급", options=["EARLY","CORE","PRE_REAL","REAL","ALL"], group=grp_entry)
use_long   = input.bool(true, "롱 허용", group=grp_entry)
use_short  = input.bool(true, "숏 허용", group=grp_entry)

// 청산 설정 (고정)
sl_pct     = input.float(1.5, "SL %", minval=0.3, maxval=5.0, step=0.1, group=grp_exit)
tp_pct     = input.float(3.0, "TP (부분익절) %", minval=0.5, maxval=10.0, step=0.1, group=grp_exit)
trail_pct  = input.float(1.0, "Trail %", minval=0.3, maxval=5.0, step=0.1, group=grp_exit)

// ATR 동적 Exit
use_atr_exit = input.bool(false, "ATR 동적 Exit", group=grp_exit)
atr_tp_mult  = input.float(3.5, "ATR TP 배수(추세)", minval=1.0, maxval=8.0, step=0.1, group=grp_exit)
atr_trail_mult = input.float(0.8, "ATR Trail 배수(추세)", minval=0.3, maxval=3.0, step=0.1, group=grp_exit)
atr_tp_floor = input.float(1.5, "TP 최소 %", minval=0.5, maxval=5.0, step=0.1, group=grp_exit)
atr_trail_floor = input.float(0.5, "Trail 최소 %", minval=0.2, maxval=3.0, step=0.1, group=grp_exit)

// 필터 설정
cooldown_bars = input.int(14, "진입 쿨다운(봉)", minval=0, maxval=50, group=grp_filt)
use_v265 = input.bool(true, "v2.6.5 sync 사용 (BNF 1H)", group=grp_filt)
score_core_min = input.int(20, "CORE 최소 score", minval=0, maxval=60, group=grp_filt)
post_core_min  = input.float(0.53, "CORE 최소 posterior", minval=0.50, maxval=0.70, step=0.01, group=grp_filt)
score_real_min = input.int(25, "REAL 최소 score", minval=0, maxval=60, group=grp_filt)

//─── 01. 기본 지표 ───

// EMA (Big Trend)
ema_fast_len = input.int(21, "EMA Fast", group=grp_filt)
ema_slow_len = input.int(55, "EMA Slow", group=grp_filt)
float ema_fast = ta.ema(close, ema_fast_len)
float ema_slow = ta.ema(close, ema_slow_len)

// Trend 판단 (15봉 EMA 일관성)
int trend_bars = 15
float confirm_ratio = 0.6
int up_cnt = 0
int down_cnt = 0
for i = 0 to trend_bars - 1
    if ema_fast[i] > ema_slow[i]
        up_cnt += 1
    if ema_fast[i] < ema_slow[i]
        down_cnt += 1
bool bull_trend = up_cnt >= math.round(trend_bars * confirm_ratio)
bool bear_trend = down_cnt >= math.round(trend_bars * confirm_ratio)
string trend_state = bull_trend ? "bull" : bear_trend ? "bear" : "neutral"

// HTF RSI (4H)
htf_rsi_len = 14
float htf_rsi = request.security(syminfo.tickerid, "240", ta.rsi(close, htf_rsi_len))
string htf_state = htf_rsi > 55 ? "bull" : htf_rsi < 45 ? "bear" : "neutral"

// Stochastic
stoch_k_len = 14
stoch_d_len = 3
stoch_smooth = 3
float k_raw = ta.stoch(close, high, low, stoch_k_len)
float k_val = ta.sma(k_raw, stoch_smooth)
float d_val = ta.sma(k_val, stoch_d_len)
bool cross_kd_up   = ta.crossover(k_val, d_val)
bool cross_kd_down = ta.crossunder(k_val, d_val)

// ADX (Regime)
adx_len = 14
float adx_val = ta.rma(math.abs(ta.change(ta.rma(ta.change(high), adx_len)) - ta.change(ta.rma(-ta.change(low), adx_len))), adx_len) / ta.rma(math.max(ta.change(high), -ta.change(low), ta.tr), adx_len) * 100
// 간소화: 방향성 지수
float di_plus_raw  = ta.rma(math.max(ta.change(high), 0), adx_len)
float di_minus_raw = ta.rma(math.max(-ta.change(low), 0), adx_len)
float atr_smooth   = ta.rma(ta.tr, adx_len)
float di_plus  = atr_smooth > 0 ? di_plus_raw / atr_smooth * 100 : 0
float di_minus = atr_smooth > 0 ? di_minus_raw / atr_smooth * 100 : 0
float dx = (di_plus + di_minus) > 0 ? math.abs(di_plus - di_minus) / (di_plus + di_minus) * 100 : 0
float adx_smooth = ta.rma(dx, adx_len)

float adx_trend_thr = 25.0
float adx_side_thr  = 18.0
string regime_state = adx_smooth >= adx_trend_thr ? "trend" : adx_smooth <= adx_side_thr ? "range" : "transition"

// Choppiness Index
int ci_length = 10
float ci_raw = 100.0 * math.log10(ta.sma(ta.atr(1), ci_length) * ci_length / (ta.highest(high, ci_length) - ta.lowest(low, ci_length) + 0.0001)) / math.log10(ci_length)
string regime_fast = ci_raw < 38.2 ? "trend" : ci_raw > 61.8 ? "range" : "transition"

// Volume
vol_sma_len = 20
float vol_sma = ta.sma(volume, vol_sma_len)
float vol_ratio = vol_sma > 0 ? volume / vol_sma : 1.0
bool vol_strong = vol_ratio >= 1.2
bool vol_ultra  = vol_ratio >= 2.0

// ATR
atr_len = 14
float atr_val = ta.atr(atr_len)
float atr_pct_val = atr_val / close * 100.0

// Ichimoku (기준선)
conv_len = 9
base_len = 26
span_b_len = 52
ichi_disp = 26
float conv_line = (ta.highest(high, conv_len) + ta.lowest(low, conv_len)) / 2
float base_line = (ta.highest(high, base_len) + ta.lowest(low, base_len)) / 2
float span_a = (conv_line + base_line) / 2
float span_b = (ta.highest(high, span_b_len) + ta.lowest(low, span_b_len)) / 2

// ADOL Proxy (HMA 기반)
adol_hma_len = 20
float adol_hma = ta.hma(close, adol_hma_len)
float adol_mom_fast = adol_hma - adol_hma[3]

// TD Sequential (간소화)
int td_buy = 0
int td_sell = 0
if close < close[4]
    td_buy := nz(td_buy[1]) + 1
else
    td_buy := 0
if close > close[4]
    td_sell := nz(td_sell[1]) + 1
else
    td_sell := 0

// Breakwater (방파제 밴드)
bw_atr_len = 14
bw_mult1 = 1.5
bw_mult2 = 3.0
float bw_atr = ta.atr(bw_atr_len)
float bw_mid = ta.sma(close, 20)
float bw_upper1 = bw_mid + bw_atr * bw_mult1
float bw_lower1 = bw_mid - bw_atr * bw_mult1

//─── 02. 스코어 계산 (Gradient Score) ───

// 컴포넌트별 strength [0.0 ~ 1.0]
f_clamp01(float x) => math.max(math.min(x, 1.0), 0.0)

float trend_sep = ema_fast > 0 ? (ema_fast - ema_slow) / ema_slow : 0.0
float trend_consist = up_cnt / float(trend_bars)
float str_trend_l = 0.5 * f_clamp01(trend_sep / 0.02) + 0.5 * trend_consist
float str_trend_s = 0.5 * f_clamp01(-trend_sep / 0.02) + 0.5 * (down_cnt / float(trend_bars))

float str_htf_l = f_clamp01((htf_rsi - 50.0) / 30.0)
float str_htf_s = f_clamp01((50.0 - htf_rsi) / 30.0)

float str_td_l = f_clamp01(td_buy / 9.0)
float str_td_s = f_clamp01(td_sell / 9.0)

float str_stoch_l = 0.6 * f_clamp01((k_val - 50.0) / 50.0) + 0.4 * f_clamp01((k_val - d_val) / 20.0)
float str_stoch_s = 0.6 * f_clamp01((50.0 - k_val) / 50.0) + 0.4 * f_clamp01((d_val - k_val) / 20.0)

float str_vol_l = f_clamp01((vol_ratio - 0.8) / (2.0 - 0.8))
float str_vol_s = str_vol_l  // 볼륨은 방향 무관

float str_regime = f_clamp01((adx_smooth - adx_side_thr) / (adx_trend_thr - adx_side_thr))

// 가중치
float w_trend  = 0.30
float w_htf    = 0.20
float w_td     = 0.18
float w_stoch  = 0.15
float w_vol    = 0.12
float w_regime = 0.05
float w_total  = w_trend + w_htf + w_td + w_stoch + w_vol + w_regime

float raw_long  = w_trend * str_trend_l + w_htf * str_htf_l + w_td * str_td_l + w_stoch * str_stoch_l + w_vol * str_vol_l + w_regime * str_regime
float raw_short = w_trend * str_trend_s + w_htf * str_htf_s + w_td * str_td_s + w_stoch * str_stoch_s + w_vol * str_vol_s + w_regime * str_regime

float score_abs_l = (raw_long / w_total) * 100.0
float score_abs_s = (raw_short / w_total) * 100.0
float score = score_abs_l - score_abs_s

//─── 03. Posterior 확률 (간소화) ───

// HTF 구조 기반 독립 prior
float htf_rsi_norm = f_clamp01((htf_rsi - 30.0) / 40.0)
float ichi_pos = close > math.max(span_a, span_b) ? 1.0 : close < math.min(span_a, span_b) ? 0.0 : 0.5
float prior_long = 0.4 * htf_rsi_norm + 0.3 * ichi_pos + 0.3 * 0.5

// Wave likelihood (Score 기반 간소화)
float score_norm = math.max(math.min(score / 100.0, 1.0), -1.0)
float wave_long = 0.5 + 0.3 * score_norm + 0.1 * (close > adol_hma ? 1.0 : -1.0) + 0.1 * (adol_mom_fast > 0 ? 1.0 : -1.0)
wave_long := math.max(math.min(wave_long, 0.95), 0.05)

// Log-odds blend
f_logit(float p) => math.log(math.max(p, 0.001) / math.max(1.0 - p, 0.001))
f_sigmoid(float x) => 1.0 / (1.0 + math.exp(-x))

float conf = math.abs(wave_long - 0.5) * 2.0
float lambda_post = 0.15 + 0.80 * conf
float logit_post = (1.0 - lambda_post) * f_logit(prior_long) + lambda_post * f_logit(wave_long)
float post_long  = f_sigmoid(logit_post)
float post_short = 1.0 - post_long

//─── 04. 위치 필터 (PATCH-54) ───

int pos_range_len = 20
float pos_high = ta.highest(high, pos_range_len)
float pos_low  = ta.lowest(low, pos_range_len)
float pos_range = pos_high - pos_low
float pos_pct = pos_range > 0 ? (close - pos_low) / pos_range : 0.5

// CORE 위치: 롱=상위20% 차단(score<40), 숏=하위20% 차단(score>-40)
bool pos_ok_core_long  = pos_pct <= 0.80 or score >= 40
bool pos_ok_core_short = pos_pct >= 0.20 or score <= -40

// REAL 위치: 롱=상위30% 차단(score<35), 숏=하위30% 차단(score>-35)
bool pos_ok_real_long  = pos_pct <= 0.70 or score >= 35
bool pos_ok_real_short = pos_pct >= 0.30 or score <= -35

// 과매수/과매도 보호
bool overbought_guard  = k_val >= 78 and score < 50
bool oversold_guard    = k_val <= 22 and score > -50

//─── 05. 신호 조건 정의 ───

// Stochastic CORE 조건
bool stoch_core_long  = (k_val <= 30) or (k_val < 70 and k_val > d_val)
bool stoch_core_short = (k_val >= 70) or (k_val > 30 and k_val < d_val)

// Stochastic override
bool stoch_override_long  = k_val <= 30
bool stoch_override_short = k_val >= 70

// === EARLY ===
bool early_soft_long = close > adol_hma and adol_hma > adol_hma[1] and adol_mom_fast > 0 and score >= 20 and trend_state != "bear" and htf_state != "bear"
bool early_soft_short = close < adol_hma and adol_hma < adol_hma[1] and adol_mom_fast < 0 and score <= -20 and trend_state != "bull" and htf_state != "bull"
bool early_long  = early_soft_long
bool early_short = early_soft_short

// === Regime OK ===
bool regime_ok_core = regime_state == "trend" or regime_state == "transition"
bool regime_ok_trend = regime_state == "trend"

// === CORE (v2.6.5 sync 기반) ===
bool base_core_long  = false
bool base_core_short = false

if use_v265
    // v2.6.5 sync: HTF + Ichimoku + Posterior + K/D
    float v265_min = 0.56
    base_core_long  := (htf_state == "bull" or trend_state == "bull") and score >= 5 and close >= base_line and post_long >= (v265_min - 0.03)
    base_core_short := (htf_state == "bear" or trend_state == "bear") and score <= -5 and close <= base_line and post_short >= (v265_min - 0.03)
else
    // 일반 CORE
    base_core_long  := bull_trend and regime_ok_core and stoch_core_long and (td_buy >= 4 or score >= score_core_min)
    base_core_short := bear_trend and regime_ok_core and stoch_core_short and (td_sell >= 4 or score <= -score_core_min)

// PATCH-53: BNF 1H 품질 강화
bool core_long_cond  = base_core_long  and score >= score_core_min and post_long >= post_core_min and pos_ok_core_long and not overbought_guard
bool core_short_cond = base_core_short and score <= -score_core_min and post_short >= post_core_min and pos_ok_core_short and not oversold_guard

// HTF 역방향 페널티 (PATCH-52)
if htf_state == "bear" and core_long_cond
    core_long_cond := score >= 35
if htf_state == "bull" and core_short_cond
    core_short_cond := score <= -35

// 횡보 레짐 차단 (PATCH-53)
if regime_state == "range"
    core_long_cond  := false
    core_short_cond := false

// === PRE_REAL (PATCH-55: 볼륨·K방향 게이트 + posterior 0.58) ===
float pre_real_score_min = 25.0
float pre_real_post_min  = 0.58
bool pre_real_long  = core_long_cond and score >= pre_real_score_min and post_long >= pre_real_post_min and vol_ratio >= 1.0 and k_val > nz(k_val[1])
bool pre_real_short = core_short_cond and score <= -pre_real_score_min and post_short >= pre_real_post_min and vol_ratio >= 1.0 and k_val < nz(k_val[1])

// HTF 필터 (PATCH-50)
if htf_state == "bear"
    pre_real_long := pre_real_long and score >= 50
if htf_state == "bull"
    pre_real_short := pre_real_short and score <= -50

// === REAL (v2.6.5 기반) ===
bool real_long_cond  = false
bool real_short_cond = false

if use_v265
    float v265_real_min = 0.57
    real_long_cond  := core_long_cond and score >= score_real_min and (vol_strong or vol_ultra) and cross_kd_up and post_long >= v265_real_min and pos_ok_real_long
    real_short_cond := core_short_cond and score <= -score_real_min and (vol_strong or vol_ultra) and cross_kd_down and post_short >= v265_real_min and pos_ok_real_short
else
    real_long_cond  := core_long_cond and regime_ok_trend and vol_strong and (td_buy >= 3 or score >= score_real_min) and pos_ok_real_long
    real_short_cond := core_short_cond and regime_ok_trend and vol_strong and (td_sell >= 3 or score <= -score_real_min) and pos_ok_real_short

// PATCH-51: 타이밍 강화 — K 전환 OR score+posterior 강한 경우 허용
bool timing_override_long  = (score >= 20 and post_long >= 0.60 and k_val <= 35 and k_val > k_val[1])
bool timing_override_short = (score <= -20 and post_short >= 0.60 and k_val >= 65 and k_val < k_val[1])
if not cross_kd_up and timing_override_long
    real_long_cond := core_long_cond and score >= score_real_min and (vol_strong or vol_ultra) and post_long >= 0.57 and pos_ok_real_long
if not cross_kd_down and timing_override_short
    real_short_cond := core_short_cond and score <= -score_real_min and (vol_strong or vol_ultra) and post_short >= 0.57 and pos_ok_real_short

//─── 06. 방향전환 쿨다운 (PATCH-54) ───

var int last_long_bar  = -999
var int last_short_bar = -999

// 쿨다운 체크
bool cooldown_ok_long  = (bar_index - last_long_bar >= cooldown_bars) and (bar_index - last_short_bar >= 3 or score >= 35)
bool cooldown_ok_short = (bar_index - last_short_bar >= cooldown_bars) and (bar_index - last_long_bar >= 3 or score <= -35)

//─── 07. 펄스 생성 (0→1 전환) ───

bool early_long_pulse     = early_long  and not early_long[1]
bool early_short_pulse    = early_short and not early_short[1]
bool core_long_pulse      = core_long_cond  and not core_long_cond[1]
bool core_short_pulse     = core_short_cond and not core_short_cond[1]
bool pre_real_long_pulse  = pre_real_long  and not pre_real_long[1]
bool pre_real_short_pulse = pre_real_short and not pre_real_short[1]
bool real_long_pulse      = real_long_cond  and not real_long_cond[1]
bool real_short_pulse     = real_short_cond and not real_short_cond[1]

// 중복 제거 (상위 등급 우선)
if real_long_pulse
    pre_real_long_pulse := false
    core_long_pulse := false
    early_long_pulse := false
if pre_real_long_pulse
    core_long_pulse := false
    early_long_pulse := false
if core_long_pulse
    early_long_pulse := false

if real_short_pulse
    pre_real_short_pulse := false
    core_short_pulse := false
    early_short_pulse := false
if pre_real_short_pulse
    core_short_pulse := false
    early_short_pulse := false
if core_short_pulse
    early_short_pulse := false

//─── 08. 진입 등급 필터 ───

bool go_long  = false
bool go_short = false

if entry_tier == "EARLY" or entry_tier == "ALL"
    if early_long_pulse and cooldown_ok_long
        go_long := true
    if early_short_pulse and cooldown_ok_short
        go_short := true
if entry_tier == "CORE" or entry_tier == "ALL"
    if core_long_pulse and cooldown_ok_long
        go_long := true
    if core_short_pulse and cooldown_ok_short
        go_short := true
if entry_tier == "PRE_REAL" or entry_tier == "ALL"
    if pre_real_long_pulse and cooldown_ok_long
        go_long := true
    if pre_real_short_pulse and cooldown_ok_short
        go_short := true
if entry_tier == "REAL" or entry_tier == "ALL"
    if real_long_pulse and cooldown_ok_long
        go_long := true
    if real_short_pulse and cooldown_ok_short
        go_short := true

// 방향 필터
if not use_long
    go_long := false
if not use_short
    go_short := false

//─── 09. Exit 규칙 (SL / TP / Trail) ───

float exit_sl    = sl_pct
float exit_tp    = tp_pct
float exit_trail = trail_pct

if use_atr_exit
    if regime_state == "trend"
        exit_tp    := math.min(math.max(atr_pct_val * atr_tp_mult, atr_tp_floor), 8.0)
        exit_trail := math.min(math.max(atr_pct_val * atr_trail_mult, atr_trail_floor), 3.0)
    else if regime_state == "range"
        exit_tp    := math.min(math.max(atr_pct_val * 2.0, atr_tp_floor), 5.0)
        exit_trail := math.min(math.max(atr_pct_val * 1.2, atr_trail_floor), 3.0)
    else
        exit_tp    := math.min(math.max(atr_pct_val * 2.8, atr_tp_floor), 6.0)
        exit_trail := math.min(math.max(atr_pct_val * 1.0, atr_trail_floor), 3.0)

//─── 10. 전략 실행 ───

if go_long
    strategy.entry("Long", strategy.long)
    strategy.exit("Long Exit", "Long",
         loss    = math.round(close * exit_sl / 100 / syminfo.mintick),
         profit  = math.round(close * exit_tp / 100 / syminfo.mintick),
         trail_offset = math.round(close * exit_trail / 100 / syminfo.mintick),
         trail_points = math.round(close * exit_tp / 100 / syminfo.mintick))
    last_long_bar := bar_index

if go_short
    strategy.entry("Short", strategy.short)
    strategy.exit("Short Exit", "Short",
         loss    = math.round(close * exit_sl / 100 / syminfo.mintick),
         profit  = math.round(close * exit_tp / 100 / syminfo.mintick),
         trail_offset = math.round(close * exit_trail / 100 / syminfo.mintick),
         trail_points = math.round(close * exit_tp / 100 / syminfo.mintick))
    last_short_bar := bar_index

// 반대 신호 시 포지션 청산
if go_short and strategy.position_size > 0
    strategy.close("Long", comment="역신호")
if go_long and strategy.position_size < 0
    strategy.close("Short", comment="역신호")

//─── 11. 시각화 ───

// 진입 신호 표시
plotshape(real_long_pulse,  style=shape.labelup, location=location.belowbar, color=color.new(color.green, 0), text="RL", size=size.small)
